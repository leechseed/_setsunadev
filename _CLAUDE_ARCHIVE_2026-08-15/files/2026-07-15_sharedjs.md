// ---------- unified storage: artifact store -> localStorage -> session memory ----------
const _mem = {};

function detectMode() {
  if (typeof window !== "undefined" && window.storage && typeof window.storage.get === "function") return "artifact";
  try {
    if (typeof window !== "undefined" && window.localStorage) {
      const k = "__dp_probe__";
      window.localStorage.setItem(k, "1");
      window.localStorage.removeItem(k);
      return "local";
    }
  } catch (e) { /* sandboxed / blocked */ }
  return "session";
}

export const STORE_MODE = detectMode();

export const store = {
  async get(key) {
    if (STORE_MODE === "artifact") {
      try { const r = await window.storage.get(key); return r ? r.value : null; }
      catch (e) { return null; }
    }
    if (STORE_MODE === "local") {
      const v = window.localStorage.getItem(key);
      return v === null ? null : v;
    }
    return key in _mem ? _mem[key] : null;
  },
  async set(key, value) {
    if (STORE_MODE === "artifact") { await window.storage.set(key, value); return; }
    if (STORE_MODE === "local") { window.localStorage.setItem(key, value); return; }
    _mem[key] = value;
  },
  async delete(key) {
    if (STORE_MODE === "artifact") { try { await window.storage.delete(key); } catch (e) {} return; }
    if (STORE_MODE === "local") { window.localStorage.removeItem(key); return; }
    delete _mem[key];
  },
};

export const STORE_LABEL = {
  artifact: "CONSOLE STORE",
  local: "THIS DEVICE",
  session: "SESSION ONLY — EXPORT BEFORE CLOSING",
}[STORE_MODE];

export const KEY = "dp:unified:v1";

export function fmt(ts) {
  const d = new Date(ts);
  return d.toLocaleString([], { month: "short", day: "numeric", hour: "2-digit", minute: "2-digit" });
}

export function esc(s) {
  return (s || "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
}
