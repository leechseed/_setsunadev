export function mcMarkdown(mc) {
  const L = [];
  TOOLS.forEach((t) => {

    if (t.note && mc.notes[t.id] && mc.notes[t.id].trim()) { L.push(""); L.push("**Notes:** " + mc.notes[t.id]); }
    if (t.synthesis) {
      const f = mc.syn.feel.filter(Boolean);
      if (f.length) { L.push("**Core feelings (ranked):**"); f.forEach((x, i) => L.push(`${i + 1}. ${x}`)); }
      if (mc.syn.delivery.trim()) { L.push(""); L.push("**Feeling → delivery:** " + mc.syn.delivery); }
      if (mc.syn.limits.trim()) { L.push(""); L.push("**Limits & edges:** " + mc.syn.limits); }
    }
    L.push("");
  });
  if (mc.ideas.length) { L.push("## Parking Lot"); mc.ideas.forEach((i) => L.push("- " + i.text)); }
  return L.join("\n");
}

function Tog({ on, onClick }) {
  return <div className={"tog" + (on ? " on" : "")} role="checkbox" aria-checked={on} onClick={onClick} />;
}

function Led({ dn, total }) {
  const cls = dn >= total && total > 0 ? "full" : dn > 0 ? "part" : "";
  const txt = dn >= total && total > 0 ? "COMPLETE" : dn > 0 ? dn + "/" + total : "STANDBY";
  return <div className={"led " + cls}><i></i><span>{txt}</span></div>;
}

function Station({ tag, name, dn, total, delay, children }) {
  return (
    <section className="station" style={{ animationDelay: (delay || 0) + "ms" }}>

  );
}
