#!/usr/bin/env bash
# ------------------------------------------------------------------
# Security Agent — Claude Code skills bootstrap
#
# Wires skills from github.com/mukul975/Anthropic-Cybersecurity-Skills
# into ./.claude/skills/ (project-scoped, so they stay contained to THIS project).
#
# NOTE: that repo is a COMMUNITY project — NOT affiliated with Anthropic.
# Skills ship with real, runnable Python (scripts/agent.py). Some domains
# (red-teaming, pentest, credential-access) are genuinely offensive. Review
# any script before letting an agent execute it. Keep Claude Code's permission
# prompts ON (do not use --dangerously-skip-permissions).
# ------------------------------------------------------------------
set -euo pipefail

REPO_URL="https://github.com/mukul975/Anthropic-Cybersecurity-Skills.git"
CACHE_DIR=".cache/cyber-skills-src"     # shallow clone lives here; delete to refresh
DEST="./.claude/skills"                 # project-scoped install target
MODE="starter"
DOMAIN=""

# --- curated defensive starter set (folder names verified against the repo) ---
STARTER=(
  # incident response
  analyzing-linux-audit-logs-for-intrusion
  analyzing-network-traffic-for-incidents
  analyzing-security-logs-with-splunk
  # threat hunting
  analyzing-persistence-mechanisms-in-linux
  analyzing-ransomware-network-indicators
  # digital forensics
  acquiring-disk-image-with-dd-and-dcfldd
  analyzing-disk-image-with-autopsy
  analyzing-memory-forensics-with-lime-and-volatility
  analyzing-browser-forensics-with-hindsight
  # soc operations
  analyzing-dns-logs-for-exfiltration
  analyzing-windows-event-logs-in-splunk
  correlating-security-events-in-qradar
  # malware analysis
  analyzing-cobalt-strike-beacon-configuration
  analyzing-bootkit-and-rootkit-samples
  # network security
  analyzing-network-traffic-with-wireshark
  analyzing-network-flow-data-with-netflow
  # cloud security
  auditing-aws-s3-bucket-permissions
  analyzing-office365-audit-logs-for-compromise
  analyzing-cloud-storage-access-patterns
  # vulnerability management
  scanning-infrastructure-with-nessus
  triaging-vulnerabilities-with-ssvc-framework
  # endpoint security
  detecting-fileless-attacks-on-endpoints
  hardening-linux-endpoint-with-cis-benchmark
  # phishing / BEC
  detecting-business-email-compromise
  analyzing-malicious-url-with-urlscan
  # ransomware defense
  detecting-ransomware-encryption-behavior
  detecting-ransomware-precursors-in-network
)

usage() {
  cat <<EOF
Security Agent — Claude Code skills bootstrap

Usage: bash setup.sh [option]

  (no option)           Install the curated defensive starter set (${#STARTER[@]} skills)
  --domain <subdomain>  Install every skill in a subdomain (e.g. digital-forensics)
  --all                 Install ALL 817 skills (heavy — see README before doing this)
  --list                List available subdomains + counts, then exit
  --dest <path>         Override install dir (default: ./.claude/skills)
  -h, --help            Show this help

Examples:
  bash setup.sh
  bash setup.sh --domain threat-hunting
  bash setup.sh --list
EOF
}

while [[ $# -gt 0 ]]; do
  case "$1" in
    --domain) MODE="domain"; DOMAIN="${2:-}"; shift 2 ;;
    --all)    MODE="all"; shift ;;
    --list)   MODE="list"; shift ;;
    --dest)   DEST="${2:-}"; shift 2 ;;
    -h|--help) usage; exit 0 ;;
    *) echo "Unknown option: $1"; echo; usage; exit 1 ;;
  esac
done

# ---- get source (clone once, then reuse; delete .cache to refresh) ----
if [[ -d "$CACHE_DIR/.git" ]]; then
  echo "> Using cached source ($CACHE_DIR). Delete it to pull a fresh copy."
else
  echo "> Cloning skill source (shallow)..."
  mkdir -p "$(dirname "$CACHE_DIR")"
  git clone --depth 1 "$REPO_URL" "$CACHE_DIR" >/dev/null 2>&1
fi
SRC="$CACHE_DIR/skills"
[[ -d "$SRC" ]] || { echo "! Source skills/ dir not found — repo layout may have changed."; exit 1; }

subdomain_of() { grep -m1 '^subdomain:' "$1" 2>/dev/null | sed 's/^subdomain:[[:space:]]*//; s/["'"'"']//g' || true; }

if [[ "$MODE" == "list" ]]; then
  echo "> Available subdomains:"
  for f in "$SRC"/*/SKILL.md; do subdomain_of "$f"; done | sort | uniq -c | sort -rn
  exit 0
fi

# ---- build install list ----
declare -a LIST=()
case "$MODE" in
  starter) LIST=("${STARTER[@]}") ;;
  all)     for d in "$SRC"/*/; do LIST+=("$(basename "$d")"); done ;;
  domain)
    [[ -z "$DOMAIN" ]] && { echo "! --domain needs a name (try: bash setup.sh --list)"; exit 1; }
    for f in "$SRC"/*/SKILL.md; do
      [[ "$(subdomain_of "$f")" == "$DOMAIN" ]] && LIST+=("$(basename "$(dirname "$f")")")
    done
    [[ ${#LIST[@]} -eq 0 ]] && { echo "! No skills for subdomain '$DOMAIN' (try: bash setup.sh --list)"; exit 1; }
    ;;
esac

# ---- copy into project ----
mkdir -p "$DEST"
installed=0; missing=0
for s in "${LIST[@]}"; do
  if [[ -d "$SRC/$s" ]]; then
    rm -rf "${DEST:?}/$s"
    cp -r "$SRC/$s" "$DEST/$s"
    installed=$((installed+1))
  else
    echo "  ! not found in repo: $s"
    missing=$((missing+1))
  fi
done

echo ""
echo "OK  Installed $installed skill(s) -> $DEST"
[[ $missing -gt 0 ]] && echo "    ($missing not found — folder names may have changed upstream)"
echo ""
echo "Next steps:"
echo "  1) Open Claude Code here:        claude"
echo "  2) Confirm skills loaded:        /skills"
echo "  3) Try one, e.g.:"
echo "       \"Use analyzing-disk-image-with-autopsy on ./evidence/disk.dd\""
echo ""
echo "Add more:   bash setup.sh --domain threat-hunting"
echo "Browse all: bash setup.sh --list"
