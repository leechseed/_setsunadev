import { useState, useEffect, useRef, useCallback } from "react";
import { createRoot } from "react-dom/client";
import { store, STORE_MODE, STORE_LABEL, KEY, LEGACY_KEY, fmt } from "./shared.js";
import MissionControl, { freshMc, mcMarkdown, doneCount, ALL_IDS, computeObjective } from "./mc.jsx";
import SexMenuProfiler, { computeResults as smpResults, buildExport as smpExport, TOTAL_ITEMS as SMP_TOTAL } from "./smp.jsx";
import RelationshipNeedsProfiler, { computeResults as rnpResults, buildYaml, modelStatement, progressUnits } from "./rnp.jsx";

// ---------- unified archive ----------
function buildUnified({ mc, smp, rnp }, opts = {}) {
  const dn = doneCount(mc), total = ALL_IDS.length, pct = Math.round((dn / total) * 100);
  const sr = smpResults(smp);
  const rp = progressUnits(rnp);
  const L = [];
  L.push("# DESIRE PROFILE — UNIFIED ARCHIVE");
  L.push("");
  L.push(`> Exported ${new Date().toLocaleString()} · Charge ${pct}% (${dn}/${total} tracker steps) · Sex Menu ${sr.rated}/${SMP_TOTAL} rated · Relationship Needs ${rp.done}/${rp.total} answered`);
  L.push("");
  L.push("---");
  L.push("");
  L.push("# PART I — MISSION CONTROL · TRACKER");
  L.push("");
  L.push(mcMarkdown(mc));
  L.push("");
  L.push("---");
  L.push("");
  L.push("# PART II — SEX MENU · SMP-1");
  L.push("");
  if (sr.rated > 0 || Object.keys(smp).length > 0) {
    L.push(smpExport(smp, sr, { brief: !!opts.brief }).replace(/^# .*\n/, ""));
  } else {
    L.push("_Not started._");
  }
  L.push("");
  L.push("---");
  L.push("");
  L.push("# PART III — RELATIONSHIP NEEDS · RNP-1");
  L.push("");
  if (rp.done > 0) {
    const r = rnpResults(rnp);
    if (rp.done < rp.total) L.push(`_Partial — ${rp.done}/${rp.total} answered. Treat scores as directional._`);
    L.push("");
    L.push("**Archetype:** " + r.archetype);
    L.push("");
    L.push("**The model:**");
    L.push("");
    modelStatement(r).forEach((p) => { L.push(p); L.push(""); });
    L.push("```yaml");
    L.push(buildYaml(r));
    L.push("```");
  } else {
    L.push("_Not started._");
  }
  L.push("");
  return L.join("\n");
}

// ---------- toast ----------
function useToast() {
  const [msg, setMsg] = useState("");
  const [show, setShow] = useState(false);
  const t = useRef(null);
  const toast = useCallback((m) => {
    setMsg(m); setShow(true);
    clearTimeout(t.current);
    t.current = setTimeout(() => setShow(false), 2600);
  }, []);
  const node = <div className={"toast" + (show ? " show" : "")}>{msg}</div>;
  return [toast, node];
}

// ---------- app ----------
function App() {
  const [phase, setPhase] = useState("boot");
  const [mc, setMc] = useState(freshMc);
  const [smp, setSmp] = useState({});
  const [rnp, setRnp] = useState({});
  const [tab, setTab] = useState("mc");
  const [saveTxt, setSaveTxt] = useState("—");
  const [lastSync, setLastSync] = useState("—");
  const [resetKey, setResetKey] = useState(0);
  const hydrated = useRef(false);
  const saveTimer = useRef(null);
  const [toast, toastNode] = useToast();

  // ---- hydrate ----
  useEffect(() => {
    (async () => {
      try {
        const raw = await store.get(KEY);
        if (raw) {
          const d = JSON.parse(raw);
          setMc({ ...freshMc(), ...(d.mc || {}), notes: { t1: "", t2: "", t3: "", t4: "", ...((d.mc || {}).notes || {}) }, syn: { feel: ["", "", "", "", ""], delivery: "", limits: "", ...((d.mc || {}).syn || {}) } });
          setSmp(d.smp || {});
          setRnp(d.rnp || {});
          if (d.updated) setLastSync(fmt(d.updated));
          setSaveTxt("loaded ✓");
        } else {
          // migrate a legacy solo-dashboard save if present
          const legacy = await store.get(LEGACY_KEY);
          if (legacy) {
            const d = JSON.parse(legacy);
            setMc({ ...freshMc(), ...d, notes: { t1: "", t2: "", t3: "", t4: "", ...(d.notes || {}) }, syn: { feel: ["", "", "", "", ""], delivery: "", limits: "", ...(d.syn || {}) } });
            setSaveTxt("migrated ✓");
          }
        }
      } catch (e) { setSaveTxt("load error"); }
      hydrated.current = true;
      setPhase("ready");
    })();
  }, []);

  // ---- autosave (debounced) ----
  useEffect(() => {
    if (!hydrated.current || phase !== "ready") return;
    setSaveTxt("saving…");
    clearTimeout(saveTimer.current);
    saveTimer.current = setTimeout(async () => {
      try {
        const updated = Date.now();
        await store.set(KEY, JSON.stringify({ v: 1, mc, smp, rnp, updated }));
        setSaveTxt("saved ✓");
        setLastSync(fmt(updated));
      } catch (e) { setSaveTxt("save error"); }
    }, 500);
    return () => clearTimeout(saveTimer.current);
  }, [mc, smp, rnp, phase]);

  // ---- stats for header/coach ----
  const sr = smpResults(smp);
  const rp = progressUnits(rnp);
  const stats = { smpRated: sr.rated, smpTotal: SMP_TOTAL, rnpDone: rp.done, rnpTotal: rp.total };
  const dn = doneCount(mc), pct = Math.round((dn / ALL_IDS.length) * 100);

  // ---- actions ----
  const copyText = async (text, okMsg) => {
    try { await navigator.clipboard.writeText(text); toast(okMsg); }
    catch { try { window.prompt("Copy this:", text); } catch (e) { toast("Copy blocked — use Export instead"); } }
  };

  const exportArchive = () => {
    const md = buildUnified({ mc, smp, rnp });
    try {
      const blob = new Blob([md], { type: "text/markdown" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url; a.download = "desire-profile-archive.md"; a.click();
      URL.revokeObjectURL(url);
      toast("Unified archive exported — one markdown, everything in it");
    } catch (e) {
      copyText(md, "Download blocked here — archive copied to clipboard instead");
    }
  };

  const copyArchive = () => copyText(buildUnified({ mc, smp, rnp }), "Full archive copied — paste it anywhere");

  const kickoff = () => {
    const obj = computeObjective(mc);
    const txt = `I'm working through my Desire Profile and want a deep-dive coaching session. Here's where I am:\n\n` +
      buildUnified({ mc, smp, rnp }, { brief: true }).replace(/^# .*\n/, "") +
      `\n\nMy current objective is: "${obj.title}". Coach me through it one step at a time — blunt, warm, frank, no hand-holding. Start by orienting me, then give me the first concrete move.`;
    copyText(txt, "Kickoff prompt copied — paste it into a Project chat");
  };

  const resetAll = async () => {
    if (!window.confirm("Wipe the whole console? This clears the tracker, the Sex Menu, the Relationship Needs profile, notes, ideas and coach history. Export the archive first if you want a copy.")) return;
    try { await store.delete(KEY); await store.delete(LEGACY_KEY); } catch (e) {}
    setMc(freshMc()); setSmp({}); setRnp({}); setResetKey((k) => k + 1);
    toast("Console reset");
  };

  if (phase === "boot") {
    return <div className="boot">BOOTING CONSOLE…</div>;
  }

  const tabs = [
    { id: "mc", label: "CONSOLE", chip: pct + "%" },
    { id: "smp", label: "SEX MENU · SMP-1", chip: sr.rated + "/" + SMP_TOTAL },
    { id: "rnp", label: "NEEDS · RNP-1", chip: rp.done + "/" + rp.total },
  ];

  return (
    <div className="wrap">
      {/* ---- header ---- */}
      <div className="top">
        <div className="brand">
          <div className="reticle"><i></i></div>
          <div>
            <h1>DESIRE PROFILE <span>// MISSION CONTROL</span></h1>
            <div className="sub">Unified Command Console · Tracker + SMP-1 + RNP-1 · Operator Eyes Only</div>
          </div>
        </div>
        <div className="meta-r">
          <div><span className={"dot" + (STORE_MODE === "session" ? " warn" : "")}></span><b>{STORE_LABEL}</b></div>
          <div style={{ marginTop: 5 }}>SAVE: <b>{saveTxt}</b></div>
          <div style={{ marginTop: 3 }}>LAST SYNC: <b>{lastSync}</b></div>
        </div>
      </div>

      {/* ---- station rail ---- */}
      <div className="tabs">
        {tabs.map((t) => (
          <button key={t.id} className={"tab" + (tab === t.id ? " on" : "")} onClick={() => setTab(t.id)}>
            <span>{t.label}</span><em>{t.chip}</em>
          </button>
        ))}
      </div>

      {/* ---- panes (kept mounted so you never lose your place) ---- */}
      <div style={{ display: tab === "mc" ? "block" : "none" }}>
        <MissionControl mc={mc} patch={setMc} stats={stats} goTab={setTab} showToast={toast} />
      </div>
      <div style={{ display: tab === "smp" ? "block" : "none" }} className="pane-form">
        <SexMenuProfiler key={"smp" + resetKey} ans={smp} setAns={setSmp} />
      </div>
      <div style={{ display: tab === "rnp" ? "block" : "none" }} className="pane-form">
        <RelationshipNeedsProfiler key={"rnp" + resetKey} ans={rnp} setAns={setRnp} />
      </div>

      {/* ---- controls ---- */}
      <div className="controls">
        <button className="btn hot" onClick={exportArchive}>⤓ Export Unified Archive (.md)</button>
        <button className="btn" onClick={copyArchive}>⎘ Copy Archive</button>
        <button className="btn" onClick={kickoff}>⎘ Copy Deep-Dive Kickoff Prompt</button>
        <button className="btn danger" onClick={resetAll}>⟲ Reset Console</button>
        <span className="privacy">🔒 Private to you · saved to: {STORE_LABEL.toLowerCase()}</span>
      </div>

      {toastNode}
    </div>
  );
}

const rootEl = document.getElementById("root");
createRoot(rootEl).render(<App />);
