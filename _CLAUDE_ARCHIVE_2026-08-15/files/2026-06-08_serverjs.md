// Desire Profile — local sync server. Zero dependencies. Run: node server.js
// Serves the dashboard and a read/write state API backed by state.json.
// Binds 127.0.0.1 only (private to this machine).

const http = require("http");
const fs   = require("fs");
const path = require("path");

const ROOT  = __dirname;
const STATE = path.join(ROOT, "state.json");
const DASH  = path.join(ROOT, "dashboard.html");
const PORT  = process.env.PORT || 8000;

const DEFAULT = {
  done: {},
  notes: { t1: "", t2: "", t3: "", t4: "" },
  quiz:  { blueprint: "", bdsmtest: "", sexualalpha: "", aellabig: "", aellaarch: "" },
  syn:   { feel: ["", "", "", "", ""], delivery: "", limits: "" },
  ideas: [],
  created: null,
  updated: null
};

function ensureState() {
  if (!fs.existsSync(STATE)) fs.writeFileSync(STATE, JSON.stringify(DEFAULT, null, 2));
}
function mtimeOf() {
  try { return Math.floor(fs.statSync(STATE).mtimeMs); } catch (e) { return 0; }
}
function readState() {
  ensureState();
  try { return JSON.parse(fs.readFileSync(STATE, "utf8")); } catch (e) { return DEFAULT; }
}
function writeStateAtomic(obj) {
  const tmp = STATE + ".tmp";
  fs.writeFileSync(tmp, JSON.stringify(obj, null, 2));
  fs.renameSync(tmp, STATE); // atomic on same filesystem
  return mtimeOf();
}
function sendJSON(res, code, obj) {
  const body = JSON.stringify(obj);
  res.writeHead(code, { "Content-Type": "application/json", "Cache-Control": "no-store" });
  res.end(body);
}

const server = http.createServer((req, res) => {
  const url = req.url.split("?")[0];

  // --- dashboard ---
  if (req.method === "GET" && (url === "/" || url === "/dashboard.html")) {
    fs.readFile(DASH, (e, data) => {
      if (e) { res.writeHead(500); res.end("dashboard.html not found in this folder"); return; }
      res.writeHead(200, { "Content-Type": "text/html; charset=utf-8" });
      res.end(data);
    });
    return;
  }

  // --- read state ---
  if (req.method === "GET" && url === "/api/state") {
    sendJSON(res, 200, { state: readState(), mtime: mtimeOf() });
    return;
  }

  // --- write state ---
  if (req.method === "POST" && url === "/api/state") {
    let body = "", tooBig = false;
    req.on("data", ch => { body += ch; if (body.length > 6_000_000) { tooBig = true; req.destroy(); } });
    req.on("end", () => {
      if (tooBig) return sendJSON(res, 413, { error: "payload too large" });
      let obj;
      try { obj = JSON.parse(body); } catch (e) { return sendJSON(res, 400, { error: "invalid json" }); }
      if (typeof obj !== "object" || Array.isArray(obj) || obj === null)
        return sendJSON(res, 400, { error: "expected a state object" });

      // conflict guard: if the file changed on disk since the client loaded it
      // (e.g. Claude Code edited it), reject and hand back the current state.
      const sent = parseInt(req.headers["if-match"] || "0", 10);
      const cur  = mtimeOf();
      if (sent && cur && cur > sent) {
        return sendJSON(res, 409, { state: readState(), mtime: cur });
      }

      const m = writeStateAtomic(obj);
      sendJSON(res, 200, { ok: true, mtime: m });
    });
    return;
  }

  res.writeHead(404, { "Content-Type": "text/plain" });
  res.end("not found");
});

ensureState();
server.listen(PORT, "127.0.0.1", () => {
